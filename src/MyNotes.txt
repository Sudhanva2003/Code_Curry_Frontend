6)Authguard.ts--------------------------------------------------------------------------------------------------------------
this file provides the foundation for the user state management(login/logout+data persistence)
throughout the app.

import {injectable} form @angular/

//this is used to perform dependency injection
//it tells angular you can make this class and inject it into other classes
//without injectable angular, will not know that it can 
//1)create this class, 2)manage its lifecycle

export interface userdata{
    userid:number,
    name:string,
    role:'user'|'Restaurant' //boolean
}
//so this typescript enforces the rules so if u assign a property which doenst mathce
//it will give compile time error.

@injectable({providedin:'root'})
//this is a very powerful line, 
//so @injectable marks this class as a service and it can be injected
//this provided in root, makes this a common instance throughout the app
//you dont have to create this instance in another file, just use this 
//to use you can just import Authguard
//and also since its defined in root you dont need to add it in the providers of other module

export class Authguard{
    private User:userdata|null=null

//so its of type userdata which we defined
//and its initialized as null
    setUser(user:userData){
        this.user=user;
        localstorage.setitem('user',json.stringify(user))
    }
    //so you already have the user variable
    //we are saying this.user to say we assigned the value
    //now in localstorage we store as key value pair
    //so we say 'user' which is the key string
    //and value is stringified(json->string) of the user object

    getUser():userData|null{
        if(!this.user){
            const stored=localstorage.getitem('user');
            if(stored)this.user=json.parse(user);
            //making it back to json from string
        }
         return this.user
    }
    //here we are seeeing we are defining the returning type its either userData or null
    //then we are checking if user is there in memory(this.user is the memory)

    //IMPORTANT: why this approach? because when u refresh the page, all the stuff in memory is lost
    getRole(): 'user' | 'restaurant' | null {
  return this.getUser()?.role ?? null;
}   //the question mark is to ensure if null wont proceed ahead.

     getId(): userId|null{
        return this.getUser()?.userid??null;
     }

     logout(){
        this.user=null;
        localstorage.removeitem(user);
     }

}
